C++（推荐）

#include <sys/prctl.h>   // 备选

// 给任意线程命名（推荐）
void set_thread_name(std::thread& t, const char* name) {
    // 必须 ≤15 字符
    pthread_setname_np(t.native_handle(), name);
}

// 当前线程自己命名
void set_current_thread_name(const char* name) {
    pthread_setname_np(pthread_self(), name);
    // 或者：prctl(PR_SET_NAME, name, 0, 0, 0);
}

Python
Python 的 threading.Thread(name=...) 默认不会设置操作系统线程名（直到较新版本才开始支持）。需要手动：
PythonCopyCopiedimport ctypes
import threading

def set_thread_name(name: str):
    name = name[:15].encode('ascii', 'replace')
    try:
        libpthread = ctypes.CDLL("libpthread.so.0")
        libpthread.pthread_setname_np.argtypes = [ctypes.c_void_p, ctypes.c_char_p]
        libpthread.pthread_setname_np.restype = ctypes.c_int
        tid = threading.current_thread().ident
        if tid is not None:
            libpthread.pthread_setname_np(tid, name)
    except Exception:
        pass

# 使用
class MyThread(threading.Thread):
    def run(self):
        set_thread_name(self.name or "worker")
        # ...

日志文件有时会用进程名（Python 进程经常显示成 python3_xxxxx.log），这也加剧了混淆。
